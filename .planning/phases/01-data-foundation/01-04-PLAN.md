**2. Create `components/BookSelector.tsx`:**

This component replaces the existing book/chapter/verse selection UI with a manifest-driven version.

Props:
```typescript
interface BookSelectorProps {
  tradition: Tradition;
  selectedBook: string | null;       // book slug
  selectedChapter: number | null;
  onBookChange: (slug: string, displayName: string) => void;
  onChapterChange: (chapter: number) => void;
}
```

Behavior:
- On mount and when `tradition` changes, call `loadManifest(tradition)` to get the book list
- Store manifest in local state via `useState` + `useEffect`
- **Book dropdown:**
  - For all traditions: render books grouped by `section` field using `<optgroup label={section}>`
  - Ethiopian Orthodox will show groups: "Old Testament", "New Testament", "Deuterocanonical", "Additional Books"
  - If no section field, render flat list
  - Label the dropdown with `NAV_LABELS[tradition].book` (shows "Surah" for Quran)
- **Chapter dropdown:**
  - When a book is selected, show chapters from 1 to `book.chapters.length`
  - Label with `NAV_LABELS[tradition].chapter`
  - For Quran: chapters.length is always 1 (each surah IS the chapter), so hide the chapter selector or auto-select chapter 1
- **Verse count display:**
  - Show "X verses" or "X ayahs" (using `NAV_LABELS[tradition].verse`) based on `book.chapters[selectedChapter - 1]`

Implementation:
```tsx
import React, { useState, useEffect } from 'react';
import { Tradition, NAV_LABELS, CanonManifest, ManifestBook } from '../services/types';
import { loadManifest } from '../services/manifestService';

interface BookSelectorProps {
  tradition: Tradition;
  selectedBook: string | null;
  selectedChapter: number | null;
  onBookChange: (slug: string, displayName: string) => void;
  onChapterChange: (chapter: number) => void;
}

export function BookSelector({
  tradition, selectedBook, selectedChapter, onBookChange, onChapterChange
}: BookSelectorProps) {
  const [manifest, setManifest] = useState<CanonManifest | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const labels = NAV_LABELS[tradition];

  useEffect(() => {
    setLoading(true);
    setError(null);
    loadManifest(tradition)
      .then(setManifest)
      .catch((err) => setError(`Failed to load ${tradition} data`))
      .finally(() => setLoading(false));
  }, [tradition]);

  if (loading) return <div>Loading...</div>;
  if (error || !manifest) return <div>{error || 'No data available'}</div>;

  const currentBook = manifest.books.find(b => b.slug === selectedBook);

  // Group books by section
  const sections = new Map<string, ManifestBook[]>();
  for (const book of manifest.books) {
    const sec = book.section || 'Books';
    if (!sections.has(sec)) sections.set(sec, []);
    sections.get(sec)!.push(book);
  }

  const isQuran = tradition === 'quran';

  return (
    <div className="book-selector">
      {/* Book / Surah selector */}
      <div>
        <label htmlFor="book-select">{labels.book}</label>
        <select
          id="book-select"
          value={selectedBook || ''}
          onChange={(e) => {
            const book = manifest.books.find(b => b.slug === e.target.value);
            if (book) {
              onBookChange(book.slug, book.displayName);
              // Auto-select chapter 1
              onChapterChange(1);
            }
          }}
        >
          <option value="">Select {labels.book}...</option>
          {Array.from(sections.entries()).map(([section, books]) => (
            <optgroup key={section} label={section}>
              {books.map(b => (
                <option key={b.slug} value={b.slug}>{b.displayName}</option>
              ))}
            </optgroup>
          ))}
        </select>
      </div>

      {/* Chapter selector — hidden for Quran (each surah = 1 chapter) */}
      {currentBook && !isQuran && currentBook.chapters.length > 1 && (
        <div>
          <label htmlFor="chapter-select">{labels.chapter}</label>
          <select
            id="chapter-select"
            value={selectedChapter || ''}
            onChange={(e) => onChapterChange(Number(e.target.value))}
          >
            {currentBook.chapters.map((_, i) => (
              <option key={i + 1} value={i + 1}>{i + 1}</option>
            ))}
          </select>
        </div>
      )}

      {/* Verse/Ayah count display */}
      {currentBook && selectedChapter && (
        <div className="verse-count">
          {currentBook.chapters[selectedChapter - 1]} {labels.verse.toLowerCase()}s
        </div>
      )}
    </div>
  );
}
```
