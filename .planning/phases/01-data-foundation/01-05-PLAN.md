```typescript
import path from 'path';
import { defineConfig, loadEnv } from 'vite';
import react from '@vitejs/plugin-react';
import { VitePWA } from 'vite-plugin-pwa';

export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, '.', '');
  return {
    server: {
      port: 3000,
      host: '0.0.0.0',
    },
    plugins: [
      react(),
      VitePWA({
        registerType: 'autoUpdate',
        devOptions: {
          enabled: false, // Do NOT enable SW in dev — prevents HMR conflicts
        },
        manifest: {
          name: 'ScriptureComix',
          short_name: 'ScriptureComix',
          description: 'Read scripture as comics — offline, free, all traditions',
          theme_color: '#1a1a2e',
          background_color: '#1a1a2e',
          display: 'standalone',
          icons: [
            {
              src: '/icon-192.png',
              sizes: '192x192',
              type: 'image/png',
            },
            {
              src: '/icon-512.png',
              sizes: '512x512',
              type: 'image/png',
              purpose: 'any maskable',
            },
          ],
        },
        workbox: {
          // ONLY cache app shell files via globPatterns — NEVER include /data/
          globPatterns: ['**/*.{js,css,html,ico,png,svg,woff,woff2}'],

          // Pre-cache canon manifests on install (per locked decision: "Canon manifests pre-cached on install")
          additionalManifestEntries: [
            { url: '/data/protestant/manifest.json', revision: null },
            { url: '/data/catholic/manifest.json', revision: null },
            { url: '/data/ethiopian/manifest.json', revision: null },
            { url: '/data/quran/manifest.json', revision: null },
          ],

          // Runtime caching: CacheFirst for all scripture data files (per-book JSON)
          runtimeCaching: [
            {
              urlPattern: /^\/data\/.*/,
              handler: 'CacheFirst',
              options: {
                cacheName: 'scripture-data-v1',
                expiration: {
                  maxEntries: 2000,  // ~550 book files + 4 manifests + headroom
                },
              },
            },
          ],
        },
      }),
    ],
    define: {
      'process.env.API_KEY': JSON.stringify(env.GEMINI_API_KEY),
      'process.env.GEMINI_API_KEY': JSON.stringify(env.GEMINI_API_KEY),
    },
    resolve: {
      alias: {
        '@': path.resolve(__dirname, '.'),
      },
    },
  };
});
```

**Critical rules:**
- `devOptions.enabled` MUST be `false` — prevents Workbox from intercepting Vite HMR websocket in development
- `globPatterns` MUST NOT include `/data/**` or `data/` — total data size is 300MB+; pre-caching would fail
- `additionalManifestEntries` MUST include all 4 manifest URLs — these are pre-cached on Service Worker install per locked decision
- `runtimeCaching` uses `CacheFirst` for `/data/**` — files are immutable scripture text, cache-first is correct
- `cacheName: 'scripture-data-v1'` — increment when data schema changes
- `maxEntries: 2000` — enough for all per-book files across all traditions

```typescript
import { describe, it, expect } from 'vitest';
import fs from 'fs';
import path from 'path';

describe('Quran data integrity', () => {
  it('Al-Fatiha verse 1 is NOT Basmala (Medina numbering)', () => {
    const filePath = path.join(process.cwd(), 'public/data/quran/yusuf-ali/al-fatiha.json');
    expect(fs.existsSync(filePath)).toBe(true);
    const data = JSON.parse(fs.readFileSync(filePath, 'utf-8'));
    // Verse 1 should be "All praise..." not "In the name of Allah..."
    const verse1 = data['1'];
    expect(verse1).toBeDefined();
    expect(verse1.toLowerCase()).not.toContain('in the name of allah');
    // Should have exactly 6 keys
    expect(Object.keys(data)).toHaveLength(6);
  });

  it('total ayah count across all 114 surahs is 6236', () => {
    const manifestPath = path.join(process.cwd(), 'public/data/quran/manifest.json');
    expect(fs.existsSync(manifestPath)).toBe(true);
    const manifest = JSON.parse(fs.readFileSync(manifestPath, 'utf-8'));
    expect(manifest.books).toHaveLength(114);
    const totalAyahs = manifest.books.reduce(
      (sum: number, book: any) => sum + book.chapters[0], 0
    );
    expect(totalAyahs).toBe(6236);
  });
});
```
